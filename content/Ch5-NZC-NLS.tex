\chapter{RH 方法求解非零边界 NLS 方程}
本章节对应范恩贵第六章, 本章节和 \ref{ch2-ZC-NLS} 有很多部分极为相似, 读者可反复对比阅读.

相比于零边界问题, 非零边界在求解渐进谱问题时除了谱参数 $k$, 对角化谱参数矩阵还会出现特征值 $\lambda$, 通常其为 $ k $ 的多值函数, 为了避免在 Riemann 面上讨论反散射的复杂性, Zakharov 和 Shabat 注意到可通过仿射变换 $ z = k + \lambda $, 其中 $ k = k(z) $, $ \lambda = \lambda(z) $ 为 $ z $ 的单值函数, 从而在 $ z $ 平面上讨论反散射或 RH 问题, 本部分和范恩贵第六章行文思路类似, 但修改符号使其更加易读.

在正式开始之前, 我们简单梳理一下求解非零边界 NLS 方程的大致思路: 如何处理复杂的渐近边界条件, 以及如何消除谱参数的多值性.

\begin{outline}[大致思路]

    考虑非零边界的 NLS 方程,
    \begin{align}
        \rmi q_{t} + q_{xx} + 2 q | q|^2 & = 0,                           \\
        \lim_{x \to \pm \infty} q(x,t)   & = q_{\pm} e^{2 \rmi q_{0}^2 t}
    \end{align}
    其中 $|q_{\pm}| = q_0 > 0$。它的 Lax 对为
    \begin{equation}
        \begin{aligned}
             & \Phi_x = U \Phi, \quad U = -\rmi k \sigma_3 + Q,                                          \\
             & \Phi_t = V \Phi, \quad V = -2\rmi k^2 \sigma_3 + \rmi \sigma_{3} (Q_{x} - Q^{2}) - 2 k Q,
        \end{aligned}
    \end{equation}
    其中 $ Q = \diag(q, -q^{*}) $。
    我们分三步将该问题转化为标准的 RH 问题:

    Step1: 为消除边界条件中关于时间 $ t $ 的依赖, 做规范变换 $ q = \hat{q} e^{2 \rmi q_{0}^{2} t}, \Phi = e^{\rmi q_{0}^{2} t \sigma_{3}} \hat{\Phi} $, 则原方程化为
    \begin{align}
        \rmi \hat{q}_{t} + \hat{q}_{xx} + 2 \hat{q} (|\hat{q}|^{2}-q_{0}^{2}) & = 0,       \\
        \lim_{x \to \pm \infty} \hat{q}(x,t)                                  & = q_{\pm},
    \end{align}
    此时边界条件已转化为与 $t$ 无关的常数, 对应的 Lax 对为
    \begin{align}
        \hat{\Phi}_{x} & = X \hat{\Phi}, \quad X = U,                             \\
        \hat{\Phi}_{t} & = T \hat{\Phi}, \quad T = V - \rmi q_{0}^{2} \sigma_{3}.
    \end{align}

    Step2: 此时虽消去 $t$, 但空间部分仍存在 $x$. 故我们取 $ x \to \pm\infty $ 的渐近状态, 此时位势矩阵变为常数矩阵, 得到渐进谱问题
    \begin{align}
        \Phi_{x} & = X_{\pm} \Phi, \quad X_{\pm} = \rmi k \sigma_{3} + Q_{\pm},                                \\
        \Phi_{t} & = T_{\pm} \Phi, \quad T_{\pm} = -2 k X_{\pm}, \quad Q_{\pm} = \diag(q_{\pm}, -q_{\pm}^{*}).
    \end{align}
    为表示方便, 我们仍然记 $ \hat{q}=q, \hat{\Phi} = \Phi$ , 但需要注意变换前后 $ q, \Phi $ 的含义不同。求解渐进矩阵 $X_{\pm}$ 的特征值, 会得到 $\pm \rmi \lambda$, 其中 $\lambda = \sqrt{k^2 + q_0^2}$.

    Step3: %引入仿射变换 $ z = k + \lambda$ (逆变换为 Joukowsky 变换), 在 Riemann 面上消去多值问题, 然后即可进入零边界问题的 RH 求解流程.
    $\lambda$ 是关于 $k$ 的多值函数, 直接在 $k$-平面上讨论解析性极为困难. 因此引入两叶 Riemann 面使其单值化,  利用仿射变换 $ z = k + \lambda$ 将 Riemann 面映射到标准的复 $z$-平面上 (其逆变换即为 Joukowsky 变换).

    至此, 即可完全套用零边界问题的 RH 求解流程.
\end{outline}
\section{非零边界的 NLS 方程}
考虑非零边界的 NLS 方程,
\begin{align}
    \rmi q_{t} + q_{xx} + 2 q | q|^2 & = 0, \label{ch5-eq-nls-orig}   \\
    \lim_{x \to \pm \infty} q(x,t)   & = q_{\pm} e^{2 \rmi q_{0}^2 t}
\end{align}
其中 $ q_{\pm}$ 与 $t$ 无关且 $ |q_{\pm}| = q_0 > 0 $. \eqref{ch5-eq-nls-orig} 的 Lax 对为,
\begin{equation} \label{ch5-eq-lax-orig}
    \begin{aligned}
         & \Phi_x = U \Phi, \quad U = -\rmi k \sigma_3 + Q,                                          \\
         & \Phi_t = V \Phi, \quad V = -2\rmi k^2 \sigma_3 + \rmi \sigma_{3} (Q_{x} - Q^{2}) - 2 k Q,
    \end{aligned}
\end{equation}
其中 $ Q = \diag(q, -q^{*}) $.

接下来我们考虑如何将边界条件转化为与 $ t $ 无关的形式, 同时为了使得渐进谱问题的时空矩阵出现线性关系, 还需对 Lax 对也做适当调整.
\begin{proposition}
    做变换
    \begin{equation}
        q \to \hat{q} e^{2 \rmi q_{0}^2 t}, \quad \Phi \to e^{\rmi q_{0}^{2} t \sigma_{3}} \hat{\Phi},
    \end{equation}
    则 NLS 方程 \eqref{ch5-eq-nls-orig} 和边界条件化为
    \begin{align}
        \rmi \hat{q}_{t} + \hat{q}_{xx} + 2 \hat{q} (|\hat{q}|^{2}-q_{0}^{2}) & = 0, \label{ch5-eq-nls-gauge}      \\
        \lim_{x \to \pm \infty} \hat{q}(x,t)                                  & = q_{\pm}, \label{ch5-eq-bc-gauge}
    \end{align}
    对应的 Lax 对为
    \begin{equation}
        \hat{\Phi}_{x} = X \hat{\Phi}, \quad \hat{\Phi}_{t} = T \hat{\Phi}, \label{ch5-eq-lax-gauge}
    \end{equation}
    其中
    \begin{equation}
        X = U = \rmi k \sigma_{3} + Q, \quad T = V - \rmi q_{0}^{2} \sigma_{3}.
    \end{equation}
\end{proposition}
\begin{remark}
    注意到 $ X = U $, $ T = V - \rmi q_{0}^{2} \sigma_{3} $, 因此变换前后 Lax 对的空间部分不变, 时间部分仅相差一个常数项, 这提醒我们在求解非零边界问题时可以直接使用变换前的空间部分, 而时间部分则需要进行适当调整.
\end{remark}
\begin{proof}
    不妨设变换 $ q \to u e^{2 \rmi \alpha t}, \Phi \to e^{\rmi \beta t \sigma_{3}} \hat{\Phi} $, 其中 $ \alpha, \beta \in \mathbb{R} $. 则原 NLS 方程 \eqref{ch5-eq-nls-orig} 化为
    \begin{equation}
        \begin{aligned}
             & u_{t} e^{2 \rmi \alpha t} + 2 \rmi \alpha u e^{2 \rmi \alpha t} + u_{xx} e^{2 \rmi \alpha t} + 2 u e^{2 \rmi \alpha t} |u e^{2 \rmi \alpha t}|^2 = 0, \\
             & \implies \rmi u_{t} + u_{xx} + 2 u (|u|^2 - \alpha) = 0.
        \end{aligned} \label{ch5-eq-nls-alpha}
    \end{equation}
    Lax 对 \eqref{ch5-eq-lax-orig} 变为
    \begin{equation}
        \begin{aligned}
            \Phi_{x}                & = e^{\rmi \beta t \sigma_{3}} \hat{\Phi}_{x} = (\rmi k \sigma_{3} + Q) e^{\rmi \beta t \sigma_{3}} \hat{\Phi}_{x}                                                                                                    \\
            \implies \hat{\Phi}_{x} & = e^{-\rmi \beta t \sigma_{3}} (\rmi k \sigma_{3} + Q) e^{\rmi \beta t \sigma_{3}} \hat{\Phi}                                                                                                                        \\
            %& = e^{-\rmi \beta t \sigma_{3}} (\rmi k \sigma_{3} + e^{\rmi \alpha t \sigma_{3}} \hat{Q} e^{-\rmi \alpha t \sigma_{3}}) e^{\rmi \beta t \sigma_{3}} \hat{\Phi}                                                       \\
                                    & = e^{\rmi (\alpha - \beta) t \sigma_{3}} \begin{pmatrix}
                                                                                   \rmi k & u \\ -u^{*} & -\rmi k
                                                                               \end{pmatrix} e^{\rmi (\beta - \alpha) t \sigma_{3}} \hat{\Phi} = e^{\rmi (\alpha - \beta) t \hat{\sigma}_{3}} \begin{pmatrix} \rmi k & u \\ -u^{*} & -\rmi k \end{pmatrix}
            %& = e^{\rmi (\alpha - \beta) t \sigma_{3}} (\rmi k \sigma_{3} + \hat{Q}) e^{\rmi (\beta - \alpha) t \sigma_{3}} \hat{\Phi} = e^{\rmi (\alpha - \beta) t \hat{\sigma}_{3}} (\rmi k \sigma_{3} + \hat{Q}) \hat{\Phi},
        \end{aligned}
    \end{equation}
    其中 $e^{\rmi (\alpha - \beta) t \hat{\sigma}_{3}} X = e^{\rmi (\alpha - \beta) t \sigma_{3}} X e^{-\rmi (\alpha - \beta) t \sigma_{3}}$
    同理有
    \begin{equation}
        \hat{\Phi} = e^{\rmi (\alpha - \beta) t \hat{\sigma}_{3}} \begin{pmatrix}
            -2 \rmi k^{2} + \rmi(|u|^{2} - \beta) & \rmi u_{x} - 2 k u                   \\
            \rmi u_{x}^{*} + 2 k u^{*}            & 2 \rmi k^{2} - \rmi(|u|^{2} - \beta)
        \end{pmatrix} \hat{\Phi}.
    \end{equation}
    取 $\alpha = \beta$, 则上述 Lax 对简化为
    \begin{align}
        \hat{\Phi}_{x} & = \begin{pmatrix}
                               \rmi k & u \\ -u^{*} & -\rmi k
                           \end{pmatrix} \hat{\Phi}= X \hat{\Phi}                                       \\
        \hat{\Phi}_{t} & = \begin{pmatrix}
                               -2 \rmi k^{2} + \rmi(|u|^{2} - \beta) & \rmi u_{x} - 2 k u                   \\
                               \rmi u_{x}^{*} + 2 k u^{*}            & 2 \rmi k^{2} - \rmi(|u|^{2} - \beta)
                           \end{pmatrix} \hat{\Phi} = T \hat{\Phi}
    \end{align} \label{ch5-eq-lax-alpha}
\end{proof}
对于方程 \eqref{ch5-eq-nls-alpha}, 其边界条件为
\begin{equation}
    \lim_{x \to \pm \infty} u(x,t) = u_{\pm}, \label{ch5-eq-bc-alpha}
\end{equation}
其中 $u_{\pm}$ 为与 $x,t$ 无关的常数, 且 $ | u_{\pm} = u_{0} > 0$, 因此 $u_{\pm} = u_{0} e^{\rmi \theta_{\pm}}$, 即两端渐近状态仅差一个相位. 基于此, 渐近 Lax 对为
\begin{align}
    \tilde{\Phi}_{x} & = \begin{pmatrix}
                             \rmi k & u_{\pm} \\ -u_{\pm}^{*} & -\rmi k
                         \end{pmatrix} \tilde{\Phi}= X \tilde{\Phi}                                           \\
    \tilde{\Phi}_{t} & = \begin{pmatrix}
                             -2 \rmi k^{2} + \rmi(|u_{0}|^{2} - \beta) & \rmi u_{x} - 2 k u_{\pm}                 \\
                             \rmi u_{x}^{*} + 2 k u_{\pm}^{*}          & 2 \rmi k^{2} - \rmi(|u_{0}|^{2} - \beta)
                         \end{pmatrix} \tilde{\Phi} = T \tilde{\Phi}
\end{align}
为使得 $\tilde{X}, \tilde{T}$ 具有线性关系, 只能取 $\beta = u_{0}^{2}$, 即证.
\section{Riemann 面和单值化坐标}
令 $ x \to \pm \infty $, 并利用边值条件 \eqref{ch5-eq-bc-gauge}, 可得
\begin{equation}
    \lim_{x\to \pm \infty} Q_{x} - Q^{2} + 2 q_{0}^{2}I = 0
\end{equation}
可得到渐进谱问题
\begin{equation}
    \Phi_{x}= X_{\pm} \Phi, \quad \Phi_{t} = T_{\pm} \Phi, \label{ch5-eq-lax-asym}
\end{equation}
其中
\begin{equation}
    X_{\pm} = \rmi k \sigma_{3} + Q_{\pm}, \quad T_{\pm} = -2 k X_{\pm}, Q_{\pm} = \diag(q_{\pm}, -q_{\pm}^{*}).
\end{equation}
对 $ X_{\pm} $ 求特征值, 可得特征方程
\begin{equation}
    |\zeta I - X_{\pm}| = (\zeta - \rmi k)(\zeta + \rmi k) + q_{0}^{2} = 0 \implies -\zeta^{2} = k^{2} + q_{0}^{2}
\end{equation}
易见 $ X_{\pm} $ 存在两个特征值 $ \zeta= \pm \rmi \lambda $, 其中 $ \lambda^{2} = k^{2}+ q_{0}^{2}$. 这里面临两个问题
\begin{enumerate}
    \item $ k \in \mathbb{C} \implies \lambda $ 为 $ k $ 的多值函数,
    \item $ k \in \mathcal{S} \implies \lambda $ 为 $ k $ 的单值函数, 但此时 $ \lambda $ 的定义域为 Riemann 面.
\end{enumerate}
在 Riemann 面上讨论反散射问题较为复杂, 为避免多值性和 Riemann 面的复杂性, 可引入一个仿射参数 $ z $, 使得从参数 $ k $ 的复平面转移到 $ z $ 的复平面上进行讨论 RH 问题.

为此, 首先考虑方程
\begin{equation}
    \lambda^{2} = k^{2} + q_{0}^{2} = (k+ \rmi q_{0})(k - \rmi q_{0}) \label{ch5-eq-lambda-k}
\end{equation}
决定的 Riemann 面, 其由支割线 $ [-\rmi q_{0}, \rmi q_{0}] $ 割开的两张 $ k$-平面 $ \mathcal{S}_{1}, \mathcal{S}_{2} $ 组成, 其中支点为 $ k = \pm \rmi q_{0} $. 在这样的 Reimann 面上, $ \lambda $ 为 $ k $ 的单值函数, 由两个单值解析分支组成, 函数值相差一个负号, 因此可以在 $ \mathcal{S}_{1}$ 上引入局部极坐标, 有
\begin{equation}
    k + \rmi q_{0} = r_{1} e^{\rmi \theta_{1}}, \quad k - \rmi q_{0} = r_{2} e^{\rmi \theta_{2}}, \quad -\pi/2 < \theta_{1}, \theta_{2} < 3\pi/2,
    \label{ch5-eq-k-polar}
\end{equation}
\begin{note}
    这里 $ q > 0 $, 则有 $ \Re(k+\rmi q_{0}) = \Re(k - \rmi q_{0}) \implies \theta_{1} = \theta_{2} $ 或 $ \theta_{1} = \theta_{2} + \pi $.
\end{note}
则可以写出 Riemann 面上的两个单值解析分支函数
\begin{equation}
    \lambda(k) = \begin{cases}
        \sqrt{r_{1} r_{2}} e^{\frac{\theta_{1} + \theta_{2}}{2}}, \quad k \in \mathcal{S}_{1}, \\
        -\sqrt{r_{1} r_{2}} e^{\frac{\theta_{1} + \theta_{2}}{2}}, \quad k \in \mathcal{S}_{2}.
    \end{cases}
\end{equation}
和
\begin{equation}
    \Im(\lambda(k)) = \begin{cases}
        \sqrt{r_{1} r_{2}} \sin\frac{\theta_{1} + \theta_{2}}{2}, \quad k \in \mathcal{S}_{1}, \\
        -\sqrt{r_{1} r_{2}} \sin\frac{\theta_{1} + \theta_{2}}{2}, \quad k \in \mathcal{S}_{2}.
    \end{cases}
\end{equation}
定义单值化变量
\begin{equation}
    z = k + \lambda,
    \label{ch5-eq-z-def}
\end{equation}
则由 \eqref{ch5-eq-lambda-k} 可得
\begin{equation}
    \lambda^{2} = k^{2} + q^{2}_{0} = (z-k)^{2} \implies q_{0}^{2} = 2 z k + z^{2} \implies k(z) = \frac{1}{2}(z - \frac{q_{0}^{2}}{z}).
    \label{ch5-eq-k-of-z}
\end{equation}
同理有 $ \lambda(z) = \frac{1}{2}(z + \frac{q_{0}^{2}}{z}) $, 由 \eqref{ch5-eq-k-polar}, 可得
\begin{align}
    \Im(k) > 0 & \implies 0 < \frac{\theta_{1} + \theta_{2}}{2} < \pi \implies \sin{\frac{\theta_{1}+\theta_{2}}{2}} > 0,    \\
    \Im(k) < 0 & \implies \pi < \frac{\theta_{1} + \theta_{2}}{2} < 2\pi \implies \sin{\frac{\theta_{1}+\theta_{2}}{2}} < 0.
\end{align}
因此, 变换 \eqref{ch5-eq-k-polar} 具有如下性质:
\begin{enumerate}
    \item 将 $\mathcal{S}_{1} $ 的 $ \Im(k)>0 $ 和 $ \mathcal{S}_{2} $ 的 $ \Im(k) < 0 $ 映射到 $ z $ 平面上 $ \Im(\lambda) > 0 $,
    \item 将 $\mathcal{S}_{1} $ 的 $ \Im(k)<0 $ 和 $ \mathcal{S}_{2} $ 的 $ \Im(k) > 0 $ 映射到 $ z $ 平面上 $ \Im(\lambda) < 0 $.
    \item 将 $ \mathcal{S}_{1} , \mathcal{S}_{2} $ 的支割线 $ [-\rmi q_{0}, \rmi q_{0}] $ 映为 $ \lambda $ 的割线 $ [-q_{0}, q_{0}] $.
\end{enumerate}
对 $ \lambda(z) = \frac{1}{2}(z + \frac{q_{0}^{2}}{z}) $ 利用 Joukowsky 变换
\begin{definition}[Joukowsky 变换]
    设 $z$ 为复平面 $\mathbb{C}$ 上的复变量, Joukowsky 变换是一个定义在去掉原点的复平面 $\mathbb{C} \setminus \{0\}$ 上的解析函数, 其标准形式定义为:
    \begin{equation}
        w = f(z) = z + \frac{1}{z}
    \end{equation}
    在更一般的流体力学应用中, 常引入一个非零实常数 $c > 0$, 其一般形式定义为:
    \begin{equation}
        w = f(z) = z + \frac{c^2}{z}
    \end{equation}
    该变换除了在临界点 $z = \pm c$ 处导数为零外, 在定义域内处处是一个共形映射.
\end{definition}

\begin{equation}
    \lambda(z) = \frac{z^{2} + q_{0}^{*}}{2 z} = \frac{|z|^{2} z + q_{0}^{2} z^{*}}{2 |z^{2}|} = \frac{(|z|^{2}-q_{0}^{2})z + q_{0}^{2}(z + z^{*})}{2 |z|^{2}} = \frac{1}{2|z|^{2}}\left[(|z|^{2}-q_{0}^{2})z + 2 q_{0}^{2} \Re(z)\right].
\end{equation}
故
\begin{equation}
    \Im(\lambda(z)) = \frac{1}{2|z|^{2}}(|z|^{2}-q_{0}^{2})\Im(z).
\end{equation}
易见 Joukowsky 变换具有如下映射性质
\begin{enumerate}
    \item 将 $ \Im(\lambda) > 0 $ 映为 $ D^{+} = \{z \in \mathbb{C}: (|z|^{2} - q_{0}^{2})\Im z > 0\}$;
    \item 将 $ \Im(\lambda) < 0 $ 映为 $ D^{-} = \{z \in \mathbb{C}: (|z|^{2} - q_{0}^{2})\Im z < 0\}$;
    \item 将割线 $ [-q_{0}, q_{0}] $ 映为圆 $ C_{0} = \{z \in \mathbb{C}: |z| = q_{0}\}$;
\end{enumerate}
对于 $ k \in \mathcal{S}_{1} $ 有
\begin{equation}
    z = k + \sqrt{k^{2}+q_{0}^{2}} = k + k(1+\frac{q_{0}^{2}}{k^{2}} + \cdots) = 2k + o(k^{-1}) \to \infty, \quad k \to \infty,
\end{equation}
对于 $ k \in \mathcal{S}_{2} $ 有
\begin{equation}
    z = k - \sqrt{k^{2}+q_{0}^{2}} = k - k(1+\frac{q_{0}^{2}}{k^{2}} + \cdots) = -\frac{q_{0}^{2}}{2k} + o(k^{-1}) \to 0, \quad k \to \infty,
\end{equation}
因此, 对于 $ k \to \infty $, $ z $ 有两种渐进状态:
\begin{equation}
    z \in \mathcal{S}_{1}: k \to \infty \implies z \to \infty, \quad z \in \mathcal{S}_{2}: k \to \infty \implies z \to 0.
\end{equation}
\section{Jost 函数性质}
\subsection{Jost 函数}
矩阵 $ X_{\pm} $ 具有两个特征值 $ \pm \rmi \lambda $, $ T_{\pm} $ 具有两个特征值 $ \mp 2\rmi k \lambda $, 且 $ X_{\pm}, T_{\pm} $ 可利用同一矩阵对角化使得
\begin{equation}
    X_{\pm} = Y_{\pm} (\rmi \lambda \sigma_{3}) Y_{\pm}^{-1}, \quad T_{\pm} = Y_{\pm} (-2 \rmi k \lambda \sigma_{3}) Y_{\pm}^{-1},
    \label{ch5-eq-xt-diag}
\end{equation}
其中
\begin{equation}
    Y_{\pm} = \begin{pmatrix}
        1                          & \frac{\rmi q_{\pm}}{z} \\
        \frac{\rmi q_{\pm}^{*}}{z} & 1
    \end{pmatrix}.
\end{equation}
\begin{proof}
    这里我们做变形 $X_{\pm} Y_{\pm} = Y_{\pm} (\rmi \lambda \sigma_{3})$ 来进行验证
    \begin{equation}
        \begin{aligned}
            LSH & = \begin{pmatrix}
                        \rmi k & q_{\pm} \\ -q_{\pm}^{*} & -\rmi k
                    \end{pmatrix}
            \begin{pmatrix}
                1                          & \frac{\rmi q_{\pm}}{z} \\
                \frac{\rmi q_{\pm}^{*}}{z} & 1
            \end{pmatrix}                                                                                                               \\
                & = \begin{pmatrix}
                        \rmi k + \frac{\rmi q_{\pm} q_{\pm}^{*}}{z}     & \frac{\rmi^{2} k q_{\pm}}{z} + q_{\pm}       \\
                        \frac{-\rmi^{2} k q_{\pm}^{*}}{z} - q_{\pm}^{*} & -\rmi k + \frac{\rmi q_{\pm} q_{\pm}^{*}}{z}
                    \end{pmatrix} \\
        \end{aligned}
    \end{equation}
    (11) 位置由 Joukowsky 变换和 \eqref{ch5-eq-k-of-z}得
    \begin{equation}
        \rmi k + \frac{\rmi q_{\pm} q_{\pm}^{*}}{z} = \rmi (k + \frac{q_{0}^{2}}{z}) = \rmi (k + \frac{q_{0}^{2}}{k+\lambda}) = \rmi \frac{k^{2} + k \lambda + q_{0}^{2}}{k+\lambda} = \rmi \frac{\lambda^{2} + k \lambda}{k+\lambda} = \rmi \lambda.
    \end{equation}
    (12) 位置元素由 \eqref{ch5-eq-z-def},
    \begin{equation}
        \frac{\rmi^{2} k q_{\pm}}{z} + q_{\pm} = q_{\pm}(1-\frac{k}{z}) = q_{\pm} \frac{\lambda}{z}
    \end{equation}
    (21), (22) 位置同理.
    \begin{equation}
        RHS = Y_{\pm} (\rmi \lambda \sigma_{3}) = \begin{pmatrix}
            \rmi \lambda                        & \frac{\rmi q_{\pm} \lambda}{z} \\
            \frac{-\rmi q_{\pm}^{*} \lambda}{z} & -\rmi \lambda
        \end{pmatrix}
    \end{equation}
\end{proof}
\begin{remark}
    在处理非零边界的 RH 问题时, 如遇到矩阵等号左右不同, 立即考虑 $z, \lambda, k, q_{0}$ 的关系, 以及 Joukowshy 变换.
\end{remark}
将 \eqref{ch5-eq-xt-diag} 带入 \eqref{ch5-eq-lax-asym}
\begin{equation}
    (Y_{\pm}^{-1} \Phi)_{x} = \rmi \lambda \sigma_{3}(Y_{\pm}^{-1} \Phi), \quad (Y_{\pm}^{-1} \Phi)_{t} = -2 \rmi k \lambda \sigma_{3}(Y_{\pm}^{-1} \Phi).
    \label{ch5-eq-lax-diag}
\end{equation}
可得渐进谱问题 \eqref{ch5-eq-lax-asym} 的解
\begin{equation}
    \Phi = Y_{\pm} E, \quad \theta(z) = \lambda(z) x - 2 k(z) \lambda(z) t.
\end{equation}
为表示方便, 我们记 $E = e^{\rmi \theta(z) \sigma}_{3}$, $\hat{E} = e^{\rmi \theta(z) \hat{\sigma}_{3}}$, 因此 Lax 对 \eqref{ch5-eq-lax-gauge} 的 Jost 函数 $ \Phi_{\pm} $ 可定义为满足如下边界条件的解
\begin{equation}
    \Phi_{\pm} \sim Y_{\pm} E, \quad x \to \pm \infty.
\end{equation}
于是对原始 Lax 对 \eqref{ch5-eq-lax-orig} 做变换
\begin{equation}
    \mu_{\pm} = \Phi_{\pm}E^{-1}, \quad x \to \pm \infty,
    \label{ch5-eq-mu-def}
\end{equation}
则
\begin{equation}
    \mu_{\pm} \sim Y_{\pm}, \quad x \to \pm \infty,
    \label{ch5-eq-mu-asymp}
\end{equation}
并得到等价 Lax 对
\begin{align}
    (Y_{\pm}^{-1} \mu_{\pm})_{x} + \rmi \lambda [Y_{\pm}^{-1}\mu_{\pm}, \sigma_{3}] = Y_{\pm}^{-1} \Delta Q_{\pm} \mu_{\pm}, \label{ch5-eq-mu-lax-x} \\
    (Y_{\pm}^{-1} \mu_{\pm})_{t} - 2 \rmi k \lambda [Y_{\pm}^{-1}\mu_{\pm}, \sigma_{3}] = Y_{\pm}^{-1} \Delta T_{\pm} \mu_{\pm}, \label{ch5-eq-mu-lax-t}
\end{align}
其中 $ \Delta Q_{\pm} = Q - Q_{\pm} $, $ \Delta T_{\pm} = T - T_{\pm} $.
\begin{proof}
    首先注意到 $ X = X_{\pm} + \Delta Q_{\pm} $, 将 \eqref{ch5-eq-mu-def} 带入 \eqref{ch5-eq-lax-gauge} 可得
    \begin{equation}
        \begin{aligned}
            (\mu_{\pm}E)_{x}                                                                         & = X (\mu_{\pm}E)                                                              \\
            \mu_{\pm, x} E + \mu_{\pm} E (\rmi \lambda \sigma_{3})                                   & = (X_{\pm} + \Delta Q_{\pm}) \mu_{\pm} E                                      \\
            \implies \mu_{\pm, x} + \mu_{\pm} (\rmi \lambda \sigma_{3})                              & = (Y_{\pm} (\rmi \lambda \sigma_{3}) Y_{\pm}^{-1} + \Delta Q_{\pm}) \mu_{\pm} \\
            \implies (Y_{\pm}^{-1} \mu_{\pm})_{x} + \rmi \lambda [Y_{\pm}^{-1}\mu_{\pm}, \sigma_{3}] & = Y_{\pm}^{-1} \Delta Q_{\pm} \mu_{\pm}.
        \end{aligned}
    \end{equation}
\end{proof}
上面两个式子可写为全微分形式
\begin{equation}
    \rmd (\hat{E}^{-1} Y_{\pm}^{-1} \mu_{\pm}) = \hat{E}^{-1} [Y
        _{\pm}^{-1} (\Delta Q_{\pm} \rmd x + \Delta T_{\pm} \rmd t) \mu_{\pm}]. \label{ch5-eq-mu-diff}
\end{equation}
对全微分形式 \eqref{ch5-eq-mu-diff} 沿特殊路径 $(-\infty, t) \to (x, t)  $ 和 $ (\infty, t) \to (x, t) $ 分别积分, 可得 Jost 函数的 Volterra 积分方程
\begin{align}
    \mu_{-}(x,t,z) & = Y_{-} + \int_{-\infty}^{x} Y_{-} e^{\rmi \lambda (x-y) \hat{\sigma}_{3}} Y_{-}^{-1} \Delta Q_{-}(y,t) \mu_{-}(y,t,z) \rmd y, \label{ch5-eq-volterra-minus} \\
    \mu_{+}(x,t,z) & = Y_{+} - \int_{x}^{\infty} Y_{+} e^{\rmi \lambda (x-y) \hat{\sigma}_{3}} Y_{+}^{-1} \Delta Q_{+}(y,t) \mu_{+}(y,t,z) \rmd y. \label{ch5-eq-volterra-plus}
\end{align}
\begin{proof}[Volterra 积分的推导]
    由 \eqref{ch5-eq-mu-diff}, 两边同时沿 $(-\infty, t) \to (x, t) $ 路径对 $ x $ 积分, 可得
    \begin{equation}
        \begin{aligned}
            \hat{E} \mu_{-}|^{x}_{-\infty}         & = \int_{-\infty}^{x} \hat{E} \Delta Q_{-} \mu_{-} \rmd y,                                                                      \\
            \hat{E} \mu_{-}(x,t,z) - \hat{E} Y_{-} & = \int_{-\infty}^{x} \hat{E} \Delta Q_{-} \mu_{-} \rmd y,                                                                      \\
            \implies \mu_{-}(x,t,z)                & = Y_{-} + \int_{-\infty}^{x} Y_{-} e^{\rmi \lambda (x-y) \hat{\sigma}_{3}} Y_{-}^{-1} \Delta Q_{-}(y,t) \mu_{-}(y,t,z) \rmd y.
        \end{aligned}
    \end{equation}
    $\mu_{+} $ 的 Volterra 积分方程同理.
\end{proof}
\subsection{$\mu_{\pm}$ 的依赖性}
由于 $ \Phi_{\pm} = \mu_{\pm} E $ 满足 Lax 对 \eqref{ch5-eq-lax-gauge}, 且其解线性相关, 因此存在与 $ x, t $ 无关的矩阵 $ S(z) $ 使得
\begin{equation}
    \Phi_{+} = \Phi_{-} S(z) \implies \mu_{+} E = \mu_{-} E S(z) \implies \mu_{+} = \mu_{-} E S(z) E^{-1}.
    \label{ch5-eq-scatter-phi}
\end{equation}
即
\begin{equation}
    \mu_{+}(x,t,z) = \mu_{-}(x,t,z) \hat{E} S(z).
    \label{ch5-eq-scatter-mu}
\end{equation}
其中 $ S(z) = (s_{ij}(z)) $ 与 $ x, t $ 无关, 称为散射矩阵.
\subsection{$\mu_{\pm}$ 和 $S(z)$ 的解析性}
\begin{theorem}
    对于 $ q(x) -q_{\pm} \in L^{1}(\mathbb{R}) $, 由 \eqref{ch5-eq-volterra-minus} 和 \eqref{ch5-eq-volterra-plus} 定义的特征函数 $ \mu_{\pm} $ 存在且唯一.
\end{theorem}
\begin{proof}
    仅证明 $\mu_{-,1}$, 对 Volterra 积分 \eqref{ch5-eq-volterra-minus} 左右两边同时乘以 $ Y_{-}^{-1} $ 可得
    \begin{equation}
        \begin{aligned}
            Y_{-}^{-1} \mu_{-}(x, t, z) & = I + \int_{-\infty}^{x} e^{\rmi \lambda(x -y)} (Y_{-}^{-1} \Delta Q_{-}(y,t) Y_{-}) (Y_{-}^{-1} \mu_{-}(y,t,z)) e^{-\rmi \lambda(x -y)}\rmd y.  \\
                                        & = I + \int_{-\infty}^{x} e^{\rmi \lambda(x -y)} (Y_{-}^{-1} \Delta X_{-}(y,t) Y_{-}) (Y_{-}^{-1} \mu_{-}(y,t,z)) e^{-\rmi \lambda(x -y)} \rmd y.
            \label{ch5-eq-volterra-minus-y}
        \end{aligned}
    \end{equation}
    不妨记 $ \mu_{\pm} = (\mu_{\pm,1}, \mu_{\pm,2}) $, 令 $ w(x,t,z) = Y_{-}^{-1} \mu_{-,1} $, 可将 \eqref{ch5-eq-volterra-minus-y} 中 $ \mu_{-,1} $ 的积分方程写为
    \begin{equation}
        w(x,t,z) = \begin{pmatrix}
            1 \\
            0
        \end{pmatrix} + \int_{-\infty}^{x} F w(y,t,z) \rmd y.
    \end{equation}
    其中
    \begin{equation}
        F = F(x-y, z) = \diag(1, e^{-2 \rmi \lambda (x-y)}) Y_{-}^{-1} \Delta X_{-}(y,t) Y_{-}.
        \label{ch5-eq-volterra-f}
    \end{equation}
    记 $ D_{-}^{\epsilon} = D_{-} \setminus \left( B_{\epsilon}(\rmi q_{0}) \cup B_{\epsilon}(- \rmi q_{0})\right) $, 其中 $ B_{\epsilon}(\pm \rmi q_{0}) = \{ z \in \mathbb{C} : |z \mp \rmi q_{0}| < \epsilon \} $, 且 $ 0 < \epsilon < 1 $. 定义递推序列
    \begin{align}
        w^{0}     & = (1, 0)^{T},                                                           \\
        w^{(n+1)} & = \int_{-\infty}^{x}F w^{(n)}(y,t,z) \rmd y, \quad n = 0, 1, 2, \cdots.
        \label{ch5-eq-volterra-iter}
    \end{align}
    由此构建 Neumann 级数
    \begin{equation}
        w(x,t,z) = \sum_{n=0}^{\infty} w^{(n)}(x,t,z).
        \label{ch5-eq-volterra-neumann}
    \end{equation}
    接下来只需证明 $ w(x, t, z) $ 为方程 \eqref{ch5-eq-volterra-minus} 在 $D_{-}^{\epsilon} $ 上的解. 仅需证明级数收敛且唯一即可.

    Step 1. $w(x,t,z)$的存在性

    定义向量的 $ L^{1} $ 范数为 $ \|w\| = |w_{1}| + |w_{2}| $, 则
    \begin{equation}
        \|w^{(n+1)}\| \leq \int_{-\infty}^{x} \|F\| \|w^{(n)}\| \rmd y,
        \label{ch5-eq-volterra-est}
    \end{equation}
    直接计算可得如下估计
    \begin{align}
        \|F \|                                   & = \| \diag(1, e^{-2 \rmi \lambda (x-y)}) \| \cdot \|Y_{-}^{-1}\| \cdot \| \Delta X_{-}(y,t) \| \cdot \| Y_{-} \|,
        \label{ch5-eq-volterra-est-f}                                                                                                                                                                                       \\
        c_{\epsilon}                             & = \max_{z \in D_{-}^{\epsilon}} \|Y_{-}^{-1}\| \cdot \| Y_{-} \| = \max_{z \in D_{-}^{\epsilon}} \frac{4(1+ q/|z|)^{2}}{1+q_{0}^{2}/z^{2}} < 4(1+\epsilon^{2})/\epsilon,
        \label{ch5-eq-volterra-est-ceps}                                                                                                                                                                                    \\
        \| \diag(1, e^{-2 \rmi \lambda (x-y)})\| & \leq 1+ e^{2\Im\lambda (x-y)}  \leq 2,
        \label{ch5-eq-volterra-est-diag}                                                                                                                                                                                    \\
        \| \Delta X_{-} \|                       & \leq 2 | q(x) - q_{-} |
        \label{ch5-eq-volterra-est-dx}
    \end{align}
    将上述估计带入 \eqref{ch5-eq-volterra-est}，可得
    \begin{equation}
        \|w^{(n+1)}(x,t,z)\| \leq 4 c_{\epsilon} \int_{-\infty}^{x} |q-q_{-}| \cdot \|w^{(n)}(y,t,z) \| \rmd y.
        \label{ch5-eq-volterra-est2}
    \end{equation}
    不妨记
    \begin{equation}
        \rho(x,t) = 4 c_{\epsilon} \int_{-\infty}^{x} |q-q_{-}| \rmd y,
    \end{equation}
    则易得当 $ q(x) - q_{-} \in L^{1}(\mathbb{R}) $ 时，对于任意固定 $x$, $ \rho(x,t) < \infty $, 积分存在, 且
    \begin{equation}
        \rho_{x}(x,t) = 4 c_{\epsilon} |q-q_{-}|, \quad w^{1} = \rho(x,t),
    \end{equation}
    由数学归纳法, 假设 $ \|w^{(n)}\| \leq \frac{\rho^{n}}{n!} $, 则由 \eqref{ch5-eq-volterra-est2} 可得
    \begin{equation}
        \begin{aligned}
            \|w^{(n+1)}\| & \leq 4 c_{\epsilon} \int_{-\infty}^{x} |q -q_{-}| \frac{\rho^{n}(y,t)}{n!} \rmd y               \\
                          & = \int_{-\infty}^{x} \rho_{x}(y,t) \frac{\rho^{n}(y,t)}{n!} \rmd y = \frac{\rho^{n+1}}{(n+1)!}.
        \end{aligned}
    \end{equation}
    由此可得 \eqref{ch5-eq-volterra-minus} 的级数解积分收敛且有界, 另外由于 $ w^{(0)}, F$ 对于 $z$ 的解析性, 递推可得 $ w^{(n)}, n \geq 1 $ 时解析.

    由于 $ q(x) - q_{-} \in L^{1}(\mathbb{R}_{-}) $, 则对 $ x \leq a \in \mathbb{R}$ 及 $ z \in D_{-}^{\epsilon} $, 有
    \begin{equation}
        \rho(x,t) \leq 4 c_{\epsilon} \int_{-\infty}^{a} |q-q_{-}| \rmd y = \sigma < \infty,
    \end{equation}
    其中 $ \sigma $ 为与 $ x$ 无关的常数, 故
    \begin{equation}
        \|w^{(n)}\| \leq \frac{\sigma^{n}}{n!},
    \end{equation}
    故 \eqref{ch5-eq-volterra-neumann} 定义的 Neumann 级数在 $ x \in (-\infty,a) $ 上绝对一致收敛, 且在 $ z \in D_{-}^{\epsilon} $ 内解析.
    利用 \eqref{ch5-eq-volterra-iter}-\eqref{ch5-eq-volterra-neumann}, 有
    \begin{equation}
        \begin{aligned}
            w(x,t,z) & = w^{(0)} + \sum_{n=1}^{\infty} w^{(n)}                                                                                                              \\
                     & = w^{(0)} + \sum_{n=1}^{\infty} \int_{-\infty}^{x} F w^{(n)}(y,t,z) \rmd y = w^{(0)} + \int_{-\infty}^{x} F \sum_{0}^{+\infty} w^{(n)}(y,t,z) \rmd y \\
                     & = w^{(0)} + \int_{-\infty}^{x} F w(y,t,z) \rmd y,
        \end{aligned}
    \end{equation}
    故 \eqref{ch5-eq-volterra-neumann} 定义的 $ w(x,t,z) $ 是方程 \eqref{ch5-eq-volterra-minus} 的解.

    Step 2. $w(x,t,z)$的唯一性:

    为证明唯一性, 不妨设 $ \tilde{w}(x,t,z) $ 也是方程 \eqref{ch5-eq-volterra-minus} 的解, 令 $ h = \tilde{w}(x,t,z) - w(x,t,z)$, 则
    \begin{equation}
        \|h(x,t,z)\| \leq \int_{-\infty}^{x} \| F \| \cdot \| h(y,t,z) \| \rmd y.
    \end{equation}
    利用 \eqref{ch5-eq-volterra-est-f}-\eqref{ch5-eq-volterra-est-dx} 估计得
    \begin{equation}
        \|h(x,t,z)\| \leq 4 c_{\epsilon} \int_{-\infty}^{x} |q-q_{-}| \cdot \|h(y,t,z)\| \rmd y = \int_{-\infty}^{x} \rho_{x}(y,t) \|h(y,t,z)\| \rmd y.
    \end{equation}
    由 Gronwall 不等式可得 $h(x,t,z) = 0$, 从而证明了 $w(x,t,z)$ 的唯一性.
    %因此，级数 \eqref{ch5-eq-volterra-minus} 在 $D_{-}^{\epsilon} $ 上收敛，从而证明了 $ w(x, t, z) $ 的存在性.

\end{proof}
由 \eqref{ch5-eq-volterra-f} 可得 $\mu_{-,1}$ 在 $D_{-}$ 上解析. 同理有 $\mu_{+,2}$ 在 $D_{-}$ 上解析, $\mu_{-,2}$ 和 $\mu_{+,1}$ 在 $D_{+}$  上解析. 即
\begin{equation}
    \mu_{+} = \begin{pmatrix}
        \mu_{+}^{(11)} & \mu_{+}^{(12)} \\
        \mu_{+}^{(21)} & \mu_{+}^{(22)}
    \end{pmatrix} = (\mu_{+,1}, \mu_{+,2}), \quad
    \mu_{-} = \begin{pmatrix}
        \mu_{-}^{(11)} & \mu_{-}^{(12)}  \\
        \mu_{-}^{(21)} & \mu_{-}^{(22)})
    \end{pmatrix} = (\mu_{-,1}, \mu_{-,2}).
\end{equation}
由 Lax 对 \eqref{ch5-eq-lax-gauge} 可得 $\tr(X) = \tr(T) = 0$, 由 Abel 定理可得
\begin{equation}
    \det(\Phi_{\pm})_{x} = \det(\Phi_{\pm})_{t} = 0 \implies \det(\Phi_{\pm}) = C,
\end{equation}
而由变换 \eqref{ch5-eq-mu-def} 可得
\begin{equation}
    \det(\mu_{\pm}) = \det(\Phi_{\pm} E) = \det(\Phi_{\pm}).
\end{equation}
故 $ \det(\mu_{\pm}) = c $ 与 $x,t $ 无关, 再利用 \eqref{ch5-eq-mu-asymp} 有
\begin{equation}
    \det(\mu_{\pm}) = \lim_{x \to \pm \infty} \det(\mu_{\pm}) = \det(Y_{\pm}) = 1+ \frac{q_{0}^{2}}{z^{2}} = \gamma \neq 0
    \label{ch5-eq-det-mu}
\end{equation}
故 $\mu_{\pm}$ 可逆, 计算可得
\begin{align}
    \mu_{+}^{-1} & = \frac{1}{\gamma} \begin{pmatrix} \mu_{+}^{(22)} & -\mu_{+}^{(12)} \\ -\mu_{+}^{(21)} & \mu_{+}^{(11)} \end{pmatrix} = (\hat{\mu}_{+}^{-}, \hat{\mu}_{+}^{+})^{T}, \\
    \mu_{-}^{-1} & = \frac{1}{\gamma} \begin{pmatrix} \mu_{-}^{(22)} & -\mu_{-}^{(12)} \\ -\mu_{-}^{(21)} & \mu_{-}^{(11)} \end{pmatrix} = (\hat{\mu}_{-}^{+}, \hat{\mu}_{-}^{-})^{T}.
\end{align}
由 \eqref{ch5-eq-scatter-mu}, \eqref{ch5-eq-det-mu} 可得 $ \det{S} = 1 $, 且
\begin{equation}
    \begin{aligned}
          & \hat{E} S(z) = \hat{E} \begin{pmatrix} s_{11} & s_{12} \\ s_{21} & s_{22} \end{pmatrix}                                                                                                                                                                                                            \\
        = & \mu_{-}^{-1} \mu_{+} = \begin{pmatrix} \hat{\mu}_{-}^{+} \\ \hat{\mu}_{-}^{-} \end{pmatrix} \begin{pmatrix} \mu_{+,1} & \mu_{+,2} \end{pmatrix} = \begin{pmatrix} \hat{\mu}_{-}^{+}\mu_{+,1} & \hat{\mu}_{-}^{+}\mu_{+,2} \\ \hat{\mu}_{-}^{-}\mu_{+,1} & \hat{\mu}_{-}^{-}\mu_{+,2} \end{pmatrix}
    \end{aligned}
\end{equation}
由此可见, $ s_{11}(z) $ 在 $D^{+}$ 上解析, $ s_{22}(z)$ 在 $D^{-}$ 上解析, $s_{12}(z), s_{21}(z)$ 连续到边界 $ \sum $.

\begin{remark}
    这里也可以用另一种方法证明 $S(z)$ 的解析性, 将 \eqref{ch5-eq-scatter-mu} 展开得
    \begin{align}
        \mu_{+,1}(z) & = s_{11}(z) \mu_{-,1}(z) + s_{21}(z) e^{-2 \rmi \theta(z)} \mu_{-,2}(z),
        \label{ch5-eq-scatter-mu1}                                                              \\
        \mu_{+,2}(z) & = s_{12}(z) e^{2 \rmi \theta(z)} \mu_{-,1}(z) + s_{22}(z) \mu_{-,2}(z),
        \label{ch5-eq-scatter-mu2}
    \end{align}
    则
    \begin{align}
        s_{11}(z) = \frac{1}{\gamma}\det(\mu_{+,1}, \mu_{-,2}), \quad            & s_{21}(z) = \frac{1}{\gamma} e^{2 \rmi \theta(z)} \det(\mu_{-,1}, \mu_{+,1}), \label{ch5-eq-scatter-s11-s21}
        \\
        s_{12}(z) = \frac{1}{\gamma} e^{-2 \rmi \theta(z)} \det(\mu_{+,2}, \quad & \mu_{-,2}), \quad s_{22}(z) = \frac{1}{\gamma}\det(\mu_{-,1}, \mu_{+,2}). \label{ch5-eq-scatter-s12-s22}
    \end{align}
    由 $ \mu_{\pm} $ 各列的解析性可知, $ s_{11}$ 在 $D^{+}$ 上解析, $s_{22}$ 在 $D^{-}$ 上解析, $s_{12}, s_{21}$ 连续到边界.
\end{remark}
\subsection{$\mu_{\pm}$ 和 $S(z)$ 的对称性}
\begin{theorem}
    $\mu_{\pm}$ 和 $S(z)$ 具有如下对称性
    \begin{align}
        \mu_{\pm}(x,t,z) & = - \sigma \mu_{\pm}^{*}(x,t,z^{*}) \sigma                           \\
        \mu_{\pm}(x,t,z) & = \frac{\rmi}{z} \mu_{\pm}(x,t, -\frac{q_{0}}{z}) \sigma_{3} Q_{\pm} \\
        S^{*}(z^{*})     & = - \sigma S(z) \sigma                                               \\
        S(z)             & = (\sigma_{3}Q_{-} )^{-1} S(-\frac{q_{0}^{2}}{z} )(\sigma_{3} Q_{+})
    \end{align}
    其中 $\sigma = \rmi \sigma_{2}$, $\sigma_{3} $ 为 Pauli 矩阵.
\end{theorem}
\begin{remark}
    这里使用 $\sigma$ 而非 $\sigma_{2}$ 是因为实矩阵无需考虑共轭以及直觉性更强. 若使用 $\sigma_{2}$ 则失去了这样的简洁性.
\end{remark}
上述对称性按照矩阵列展开为
\begin{align}
    \mu_{\pm,1}(x,t,z)                                           & = \sigma \mu_{\pm,2}^{*}(x,t,z^{*})
    \label{ch5-eq-sym-mu1}                                                                                                             \\
    \mu_{\pm,2}(x,t,z)                                           & = - \sigma \mu_{\pm,1}^{*}(x,t,z^{*})
    \label{ch5-eq-sym-mu2}                                                                                                             \\
    \mu_{\pm,1}(x,t,z)                                           & = \frac{\rmi q^{*}}{z} \mu_{\pm,2}(x,t, -\frac{q_{0}}{z})
    \label{ch5-eq-sym-mu3}                                                                                                             \\
    \mu_{\pm,2}(x,t,z)                                           & = - \frac{\rmi q}{z} \mu_{\pm,1}(x,t, -\frac{q_{0}}{z})
    \label{ch5-eq-sym-mu4}                                                                                                             \\
    s_{11}(z) = s_{22}^{*}(z^{*}),                               & \quad s_{12}(z) = - s_{21}^{*}(z^{*}),
    \label{ch5-eq-sym-scatter}                                                                                                         \\
    s_{11}(z) = \frac{q_{+}}{q_{-}}s_{22}(-\frac{q_{0}^{2}}{z}), & \quad s_{12}(z) = \frac{q_{+}}{q_{-}} s_{21}(-\frac{q_{0}^{2}}{z}).
    \label{ch5-eq-sym-scatter2}
\end{align}


\begin{proof}
    由
    \begin{equation}
        \left[ Y_{\pm}^{-1} \mu_{\pm}(z) \right]_{x} + \rmi \lambda \left[ Y_{\pm}^{-1}(z) \mu_{\pm}(z), \sigma_{3} \right] = Y_{\pm}^{-1}(z) \Delta{Q_{\pm}(z)} \mu_{\pm}(z)
        \label{ch5-eq-sym-ode}
    \end{equation}
    将 $ z \to z^{*} $, 再对方程两边同时共轭得
    \begin{equation}
        \left[Y_{\pm}^{-1}(z) \mu_{\pm}(z^{*}) \right]_{x}^{*} - \rmi \lambda^{*}(z^{*}) \left[ Y_{\pm}^{-1} \mu_{\pm}^{*}(z^{*}), \sigma_{3} \right] = Y_{\pm}^{-1*}(z^{*}) \Delta{Q_{\pm}^{*}(z^{*})} \mu_{\pm}^{*}(z^{*})
        \label{ch5-eq-sym-conj}
    \end{equation}
    对两边同时乘以 $ \sigma $ 得
    \begin{equation}
        \sigma \left[Y_{\pm}^{-1}(z) \mu_{\pm}(z^{*}) \right]_{x}^{*} \sigma - \rmi \lambda^{*}(z^{*}) \sigma \left[ Y_{\pm}^{-1} \mu_{\pm}^{*}(z^{*}), \sigma_{3} \right] \sigma = \sigma Y_{\pm}^{-1*}(z^{*}) \Delta{Q_{\pm}^{*}(z^{*})} \mu_{\pm}^{*}(z^{*}) \sigma
        \label{ch5-eq-sym-sigma}
    \end{equation}
    由 $ Y_{\pm}, \Delta{Q_{\pm}}, \lambda $ 定义, 注意到
    \begin{equation}
        \sigma Y_{\pm}^{-1*}(z^{*}) \sigma = - Y_{\pm}^{-1}(z), \quad \lambda^{*}(z^{*}) = \lambda(z), \quad \sigma \Delta Q_{\pm}^{*}(z^{*}) \sigma = - \Delta Q_{\pm}(z).
        \label{ch5-eq-sym-id}
    \end{equation}
    则有
    \begin{equation}
        \left[ Y_{\pm}^{-1}(z) \sigma \mu_{\pm}^{*}(z^{*}) \sigma \right]_{x} + \rmi \lambda(z) \left[  Y_{\pm}^{-1}(z) \sigma \mu_{\pm}^{*}(z^{*}) \sigma, \sigma_{3} \right] = Y_{\pm}^{-1}(z) \Delta{Q_{\pm}(z)} \sigma \mu_{\pm}^{*}(z^{*}) \sigma
        \label{ch5-eq-sym-ode2}
    \end{equation}
    由 \eqref{ch5-eq-sym-ode}, \eqref{ch5-eq-sym-ode2} 可得 $ \mu_{\pm}(x,t,z) $ 和 $ -\sigma \mu_{\pm}^{*}(x,t,z^{*}) \sigma $ 都满足同一线性齐次微分方程族, 且具有相同的渐近性
    \begin{equation}
        \mu_{\pm}(x,t,z) \sim Y_{\pm}, \quad -\sigma \mu_{\pm}^{*}(x,t,z^{*}) \sigma \sim -\sigma Y_{\pm}^{*} \sigma = Y_{\pm}, \quad x \to \pm \infty,
        \label{ch5-eq-sym-asymp}
    \end{equation}
    故二者相等, 即
    \begin{equation}
        \mu_{\pm}(x,t,z) = -\sigma \mu_{\pm}^{*}(x,t,z^{*}) \sigma.
        \label{ch5-eq-sym-mu}
    \end{equation}
    接下来考虑 $ S(z) $ 的对称性, 将 \eqref{ch5-eq-scatter-phi} 改写为
    \begin{equation}
        S(z) = \hat{E}^{-1} \mu_{-}^{-1}(x,t,z) \mu_{+}(x,t,z),
    \end{equation}
    注意到 $ \theta^{*}(z^{*}) = \theta(z) $, $ \sigma E^{*}(z^{*}) \sigma = - E $, 并利用 \eqref{ch5-eq-sym-asymp} 可得
    \begin{equation}
        \begin{aligned}
            -\sigma S^{*}(z^{*}) \sigma & = \sigma \left( E^{-1}(z^{*}) \mu_{-}^{-1}(z^{*} E(z^{*})) \mu_{+}(z^{*})\right)^{*} \sigma                                                                                                     \\
                                        & = \left( -\sigma (E^{-1}(z^{*}))^{*} \sigma \right) \left( \sigma \mu_{-}^{-1}(z^{*}) \sigma \right) \left( -\sigma \mu_{+}^{*}(z^{*})\sigma \right) \left( -\sigma E^{*}(z^{*}) \sigma \right) \\
                                        & = E^{-1} \mu_{-}^{-1} \mu_{+} E = \hat{E}^{-1} \mu_{-}^{-1} \mu_{+} = S(z)
        \end{aligned}
    \end{equation}
\end{proof}
\subsection{$\mu_{\pm}$ 和 $S(z)$ 的渐近性}
由 Lax 对 \eqref{ch5-eq-lax-orig} 及渐近替换 $\Phi = \mu E$ 出发，易见
\begin{align}
    \mu_{\pm} & = I + \frac{\rmi}{z} \sigma_{3} Q + \mathcal{O}(z^{-2}), \quad z \to \infty. \label{ch5-eq-mu-zinf} \\
    \mu_{\pm} & = \frac{\rmi}{z} \sigma_{3} Q_{\pm} + \mathcal{O}(1), \quad z \to 0. \label{ch5-eq-mu-z0}
\end{align}
更进一步可得
\begin{equation}
    S(z) = I + \mathcal{O}(z^{-1}), \quad z \to \infty, \quad S(z) = \diag(\frac{q_{-}}{q_{+}}, \frac{q_{+}}{q_{-}}) + \mathcal{O}(z), \quad z \to 0.
    \label{ch5-eq-S-asymptotic}
\end{equation}

\begin{proof}[这里书上证明过于繁琐, 我给出一个更简洁地证明]
    仅证明 $z \to \infty$ 时的渐近行为, 考虑空间部分的 Lax 方程. 将 $\Phi = \mu_{\pm} E$ 代入 $\Phi_{x} = (\rmi k \sigma_{3} + Q)\Phi$, 两边右乘 $E^{-1}$
    \begin{equation}
        \mu_{\pm, x} - \rmi k \sigma_{3} \mu_{\pm} + \rmi \lambda \mu_{\pm} \sigma_{3} = Q \mu_{\pm}.
        \label{ch5-eq-mu-ode-simple}
    \end{equation}

    由单值化参数定义 $z = k + \lambda$ 及 $\lambda^2 = k^2 + q_0^2$ 可知, 当 $z \to \infty$ 时:
    \begin{equation}
        \lambda = \frac{z}{2} + \frac{q_0^2}{2z}, \quad k = \frac{z}{2} - \frac{q_0^2}{2z}.
    \end{equation}
    $\mu_{\pm}$ 在 $z \to \infty$ 处的渐进展开为:
    \begin{equation}
        \mu_{\pm} = \mu_{0}^{\pm} + \frac{\mu_{1}^{\pm}}{z} + \mathcal{O}(z^{-2}).
        \label{ch5-eq-mu-exp}
    \end{equation}
    将展开式带入方程 \eqref{ch5-eq-mu-ode-simple} 中，并按 $z$ 的幂次收集各项：
    \begin{equation}
        \left( \mu_{0,x}^{\pm} + \frac{\mu_{1,x}^{\pm}}{z} \right) - \rmi \left( \frac{z}{2} - \frac{q_0^2}{2z} \right) \sigma_{3} \left( \mu_{0}^{\pm} + \frac{\mu_{1}^{\pm}}{z} \right) + \rmi \left( \frac{z}{2} + \frac{q_0^2}{2z} \right) \left( \mu_{0}^{\pm} + \frac{\mu_{1}^{\pm}}{z} \right) \sigma_{3} = Q \left( \mu_{0}^{\pm} + \frac{\mu_{1}^{\pm}}{z} \right).
    \end{equation}
    比较等式两端 $z$ 的各次幂系数：
    \begin{align}
        z^{1} : & \quad \frac{\rmi}{2} [\mu_{0}^{\pm}, \sigma_{3}] = 0, \label{ch5-eq-exp-z1}                                 \\
        z^{0} : & \quad \mu_{0,x}^{\pm} + \frac{\rmi}{2} [\mu_{1}^{\pm}, \sigma_{3}] = Q \mu_{0}^{\pm}. \label{ch5-eq-exp-z0}
    \end{align}

    由 \eqref{ch5-eq-exp-z1} 可知，$\mu_{0}^{\pm}$ 为对角矩阵, 故其 $\mu_{0,x}^{\pm}$ 也是对角阵. 又由于 $[\mu_{1}^{\pm}, \sigma_{3}]$ 的对角线元素为 $0$, 同时位势矩阵 $Q$ 纯为非对角阵, 故 $Q\mu_{0}^{\pm}$ 也只有非对角元素. 分离 \eqref{ch5-eq-exp-z0} 的对角部分与非对角部分
    \begin{equation}
        \mu_{0,x}^{\pm} = 0, \quad
        \frac{\rmi}{2} [\mu_{1}^{\pm}, \sigma_{3}] = Q \mu_{0}^{\pm}. \label{ch5-eq-offdiag}
    \end{equation}
    又 $\mu_{\pm} \to I (x \to \pm\infty)$，且 $\mu_{0,x}^{\pm}=0$，故在全空间恒有 $\mu_{0}^{\pm} = I$, 代入 \eqref{ch5-eq-offdiag}，得：
    \begin{equation}
        \frac{\rmi}{2} [\mu_{1}^{\pm}, \sigma_{3}] = Q.
    \end{equation}
    易知微扰项 $\mu_{1}^{\pm}$ 对角元素为 $0$, 又 $\sigma_{3} \mu_{1}^{\pm} = -\mu_{1}^{\pm} \sigma_{3}$, 故 $[\mu_{1}^{\pm}, \sigma_{3}] = 2\mu_{1}^{\pm} \sigma_{3}$. 代入上式解得：
    \begin{equation}
        \rmi \mu_{1}^{\pm} \sigma_{3} = Q \implies \mu_{1}^{\pm} = -\rmi Q \sigma_{3} = \rmi \sigma_{3} Q.
    \end{equation}
    至此 \eqref{ch5-eq-mu-zinf} 证明完毕。进一步比较 $(1,2)$ 矩阵元，即可得到重构公式：
    \begin{equation}
        q(x,t) = -\rmi (\mu_{1}^{\pm})_{12} = -\rmi \lim_{z \to \infty} z (\mu_{\pm})_{12}.
    \end{equation}
\end{proof}
\section{相关广义 RH 问题}
由 \eqref{ch5-eq-scatter-mu1} 和 \eqref{ch5-eq-scatter-mu2} 可定义分片解析矩阵
\begin{equation}
    M^{+} = (\frac{\mu_{+,1}}{s_{11}}, \mu_{-,2}), \quad M^{-} = (\mu_{-,1}, \frac{\mu_{+,2}}{s_{22}}).
\end{equation}
易见 $M^{\pm}$ 分别在 $D^{\pm}$ 上解析, 利用 Jost 函数和散射系数的渐进性, 容易证明
\begin{equation}
    M^{\pm} = I + O(z^{-1}), \quad z \to \infty,
    \quad M^{\pm} = \frac{1}{z} \sigma_{3} Q_{\pm} + O(1), \quad z \to 0.
\end{equation}
由此我们可得到下面广义 RH 问题
\begin{theorem}
    $ M^{\pm}(x,t,z) $ 在 $ \mathbb{C} \setminus \sum $ 上亚纯, 在零点集 $ \{z : s_{11}(z) = s_{22}(z) = 0 \} $ 上满足留数条件, 且满足如下跳跃条件
    \begin{align}
        M^{-}(x,t,z) & = M^{+}(x,t,z) (I - G(x,t,z)), z \in \sum \label{ch5-eq-rh-jump} \\
        M^{\pm}      & = I + O(z^{-1}), z \to \infty                                    \\
        M^{\pm}      & = \frac{\rmi}{z} \sigma_{3} Q_{\pm} + O(1), z \to 0
    \end{align}
    其中跳跃矩阵为
    \begin{equation}
        G(x,t,z) = \hat{E} \begin{pmatrix} 0 & -\tilde{\rho}(z) \\ \rho(z) & \rho(z) \tilde{\rho}(z) \end{pmatrix},
    \end{equation}
    这里反射系数为 $ \rho(z) = s_{21}/s_{11}, \tilde{\rho}(z) = s_{12}/s_{22} $, 则 NLS 方程解可表示为
    \begin{equation}
        q(x,t) = -\rmi \lim_{z \to \infty} (z M_{+})_{12}.
    \end{equation}
\end{theorem}
\begin{proof}[跳跃矩阵 $G(x,t,z)$ 的推导]
    由分片解析矩阵 $ M^{\pm} $ 的定义可知, 它们的列向量分别为:
    \begin{equation}
        M^{+}_{1}  = \frac{\mu_{+,1}}{s_{11}}, \quad M^{+}_{2} = \mu_{-,2}, \quad M^{-}_{1} = \mu_{-,1}, \quad M^{-}_{2} = \frac{\mu_{+,2}}{s_{22}}.
    \end{equation}
    则有
    \begin{equation}
        M_{1}^{+} = M_{1}^{-} + \rho e^{-2 \rmi\theta} M_{2}^{+}, \quad M_{2}^{-} = \tilde{\rho} e^{2i\theta} M_{1}^{-} + M_{2}^{+}
        \label{ch5-eq-M-minus-1}
    \end{equation}
    将散射关系 \eqref{ch5-eq-scatter-mu1}-\eqref{ch5-eq-scatter-mu2} 两边分别同除 $s_{11}, s_{22}$, 并利用反射系数的定义得:
    \begin{align}
        \frac{\mu_{+,1}}{s_{11}} & = \mu_{-,1} + \frac{s_{21}}{s_{11}} e^{-2 \rmi \theta} \mu_{-,2} \implies M^{+}_{1} = M^{-}_{1} + \rho e^{-2 \rmi \theta} M^{+}_{2}.
        \label{ch5-eq-M-col1}                                                                                                                                                 \\
        \frac{\mu_{+,2}}{s_{22}} & = \frac{s_{12}}{s_{22}} e^{2 \rmi \theta} \mu_{-,1} + \mu_{-,2} \implies M^{-}_{2} = \tilde{\rho} e^{2 \rmi \theta} M^{-}_{1} + M^{+}_{2}.
        \label{ch5-eq-M-col2}
    \end{align}
    将 \eqref{ch5-eq-M-minus-1} 代入 \eqref{ch5-eq-M-col2} 中, 以消去 $M^{-}_{1}$:
    \begin{equation}
        \begin{aligned}
            M^{-}_{2} & = \tilde{\rho} e^{2 \rmi \theta} \left( M^{+}_{1} - \rho e^{-2 \rmi \theta} M^{+}_{2} \right) + M^{+}_{2}
            = \tilde{\rho} e^{2 \rmi \theta} M^{+}_{1} - \rho \tilde{\rho} M^{+}_{2} + M^{+}_{2}                                  \\
                      & = \tilde{\rho} e^{2 \rmi \theta} M^{+}_{1} + (1 - \rho \tilde{\rho}) M^{+}_{2}.
        \end{aligned}
        \label{ch5-eq-M-minus-2}
    \end{equation}
    联立 \eqref{ch5-eq-M-minus-1} 和 \eqref{ch5-eq-M-minus-2}:
    \begin{equation}
        M^{-} = (M^{-}_{1}, M^{-}_{2}) = (M^{+}_{1}, M^{+}_{2})
        \begin{pmatrix}
            1                        & \tilde{\rho} e^{2 \rmi \theta} \\
            -\rho e^{-2 \rmi \theta} & 1 - \rho \tilde{\rho}
        \end{pmatrix}.
    \end{equation}
    则
    \begin{equation}
        G(x,t,z)  = \begin{pmatrix} e^{\rmi \theta} & 0 \\ 0 & e^{-\rmi \theta} \end{pmatrix}
        \begin{pmatrix} 0 & -\tilde{\rho}(z) \\ \rho(z) & \rho(z) \tilde{\rho}(z) \end{pmatrix}
        \begin{pmatrix} e^{-\rmi \theta} & 0 \\ 0 & e^{\rmi \theta} \end{pmatrix}
        = \hat{E} \begin{pmatrix} 0 & -\tilde{\rho}(z) \\ \rho(z) & \rho(z) \tilde{\rho}(z) \end{pmatrix}.
    \end{equation}
\end{proof}

\begin{equation}
    \mu_{+,2}(\xi_{n}^{*}) = d_{n} e^{2\rmi \theta(\xi_{n}^{*})} \mu_{-,1}(\xi_{n}^{*}), \quad n = 1, 2, \cdots, 2 N,
    \label{ch5-eq-mu22-mu11}
\end{equation}
其中 $d_{n}$ 为与 $x, t$ 无关但 $b_n$ 有关的常数. 对上式两边同时取复共轭, 并在左乘 $\sigma$ 可得:
\begin{equation}
    \sigma \mu_{+,2}^{*}(\xi_{n}^{*}) = d_{n}^{*} e^{-2 \rmi \theta^{*}(\xi_{n}^{*})} \sigma \mu_{-,1}^{*}(\xi_{n}^{*}).
\end{equation}
利用对称性 \eqref{ch5-eq-sym-mu1} 和 \eqref{ch5-eq-sym-mu2}, 且注意到 $\theta^{*}(z^{*}) = \theta(z)$, 上式可化为:
\begin{equation}
    \mu_{+, 1}(\xi_{n}) = -d_{n}^{*} e^{-2 \rmi \theta(\xi_{n})} \mu_{-, 2}(\xi_{n}).
\end{equation}
将其与 \eqref{ch5-eq-mu11-mu22} 比较, 可得: $d_{n}^{*} = -b_{n}$. 由此计算 $\mu_{+,2}/s_{22}$ 在 $\xi_n^*$ 处的留数:

    \Res_{z = \xi_{n}^{*}} \frac{\mu_{+, 2}(z)}{s_{22}(z)} = \frac{\mu_{+, 2}(\xi_{n}^{*})}{s_{22}'(\xi_{n}^{*})} = D_{n} e^{2\rmi \theta(\xi_{n}^{*})} \mu_{-,1}(\xi_{n}^{*}), \quad D_{n} = \frac{d_{n}}{s_{22}'(\xi_{n}^{*})}.
    \label{ch5-eq-mu22-res}
\end{equation}
利用散射系数的对称性 \eqref{ch5-eq-sym-scatter} 有
\begin{equation}
    D_{n}^{*} = \left( \frac{d_{n}}{s_{22}'(\xi_{n}^{*})} \right)^{*} = \frac{d_{n}^{*}}{(s_{22}'(\xi_{n}^{*}))^{*}} = \frac{-b_{n}}{s_{11}'(\xi_{n})} = -C_{n} \implies D_{n} = -C_{n}^{*}.
\end{equation}
联立 \eqref{ch5-eq-mu11-res} 和 \eqref{ch5-eq-mu22-res} 及 $D_n = -C_n^*$, 可得分片解析矩阵 $M^{\pm}$ 的留数条件:

                                         \end{pmatrix}
    \end{aligned}
\end{equation}
\section{RH 问题的可解性}
\subsection{重构公式}
为利用 Plemelj 公式求解 RH 问题, 我们需要去除 $M$ 的奇性, 将原来的 RH 问题转化为规范 RH 问题, 将 \eqref{ch5-eq-rh-jump} 改写为
\begin{equation}
    \begin{aligned}
          & M^{-} - I - \frac{\rmi}{z} \sigma_{3} Q_{-} - \sum_{n=1}^{2N} \frac{\Res_{z = \xi_{n}^{*}} M^{-}}{z - \xi_{n}^{*}} - \sum_{n=1}^{2N} \frac{\Res_{z=\xi_{n}} M^{+}}{z - \xi_{n}}            \\
        = & M^{+} - I - \frac{\rmi}{z} \sigma_{3} Q_{-} - \sum_{n=1}^{2N} \frac{\Res_{z = \xi_{n}^{*}} M^{-}}{z - \xi_{n}^{*}} - \sum_{n=1}^{2N} \frac{\Res_{z=\xi_{n}} M^{+}}{z - \xi_{n}} - M^{+} G.
    \end{aligned}
    \label{ch5-eq-mu-plemelj}
\end{equation}
上述方程左边前四项在 $D^{-}$ 上解析, 且第五项极点为 $z = \xi \in D^{+}$, 因此在 $D^{-}$ 上解析. 于是 \eqref{ch5-eq-mu-plemelj} 左边在 $D^{-}$ 上解析且渐近于 $O(1/z), (z \to \infty)$. 同理 \eqref{ch5-eq-mu-plemelj} 右边前五项 $D^{+}$ 上解析, 且 $O(1/z), (z \to \infty)$. 于是由 Plemelj 公式可得
\begin{equation}
    M(x,t,z) = I + \frac{\rmi}{z} \sigma_{3} Q_{-} + \sum_{n=1}^{2N} \frac{\Res_{z = \xi_{n}^{*}} M^{-}}{z - \xi_{n}^{*}} + \sum_{n=1}^{2N} \frac{\Res_{z=\xi_{n}} M^{+}}{z - \xi_{n}} + \frac{1}{2\pi \rmi} \int_{\sum} \frac{M(s) G(x,t,s)}{s - z} \rmd s. \quad z \in \mathbb{C} \setminus \sum
    \label{ch5-eq-mu-plemelj-sol}
\end{equation}
\begin{remark}
    这里是因为若记 \eqref{ch5-eq-mu-plemelj} 的左边为 $F^{-}$, 右边为 $F^{+}-M^{+}G$, 则
    \begin{equation}
        F^{+}(z) - F^{-}(z) = M^{+}G, \quad \lim_{z \to \infty} F^{\pm} = 0
    \end{equation}
    满足标准 Plemelj 公式.
\end{remark}
分别在 $D^{+}$ 和 $D^{-}$ 上考虑 \eqref{ch5-eq-mu-plemelj-sol} 的第二, 第一列:

函数在点 $ z = \xi_{n} \in D^{+} $ 上, 考虑 $M^{+}$ 的 12 位置元素, 有
\begin{equation}
    \mu_{-, 2} = \begin{pmatrix}
        \frac{\rmi q_{-}}{\xi_{n}} \\ 1
    \end{pmatrix}
    - \sum_{k=1}^{2N} \frac{C_{k}^{*} e^{\rmi \theta(\xi_{k}^{*})} \mu_{-, 1}(\xi_{k}^{*})}{\xi_{n} - \xi_{k}^{*}} + \frac{1}{2\pi \rmi} \int_{\sum} \frac{(M^{+} G)_{2}}{s - \xi_{n}} \rmd s, \quad n = 1, 2, \cdots, 2 N.
    \label{ch5-eq-mu-12}
\end{equation}
函数在点 $ z = \xi_{n}^{*} \in D^{-} $ 上, 考虑 $M^{-}$ 的 21 位置元素, 有
\begin{equation}
    \mu_{-, 1} = \begin{pmatrix}
        1 \\ -\frac{\rmi q_{-}}{\xi_{n}^{*}}
    \end{pmatrix}
    + \sum_{k=1}^{2N} \frac{C_{k} e^{\rmi \theta(\xi_{k})} \mu_{-, 2}(\xi_{k})}{\xi_{n}^{*} - \xi_{k}} + \frac{1}{2\pi \rmi} \int_{\sum} \frac{(M^{-} G)_{1}}{s - \xi_{n}^{*}} \rmd s, \quad n = 1, 2, \cdots, 2 N.
    \label{ch5-eq-mu-21}
\end{equation}
故 $z \to \infty$ 时, \eqref{ch5-eq-mu-plemelj-sol} 的渐近展开为
\begin{equation}
    M(x,t,z) = I + \frac{1}{z} \left\{ \rmi \sigma_{3} Q_{-} + \sum_{n=1}^{2N} \left( \Res_{\xi_{n}} M^{+} + \Res_{\xi_{n}^{*}} M^{-} \right) - \frac{1}{2\pi \rmi} \int_{\sum} M^{+}(s) G(s) \rmd s \right\} + O(z^{-2}).
    \label{ch5-eq-mu-zinf-plemelj}
\end{equation}
取 $M = M^{+}$, 比较矩阵 \eqref{ch5-eq-mu-zinf-plemelj} 和 \eqref{ch5-eq-mu-zinf} 的 12 位置元素, 可得 NLS 方程位势函数和 RH 问题解相联系的重构公式
\begin{equation}
    q(x,t) = -\rmi \lim_{z \to \infty} z (M^{+})_{12} = q_{-} + \sum_{n=1}^{2N} C_{n}^{*} e^{\rmi \theta(\xi_{n}^{*})} \mu_{-,1,1}^{(1)}(\xi_{n}^{*}) + \frac{1}{2\pi} \int_{\sum} (M^{+}(s) G(s))_{12} \rmd s.
\end{equation}
\subsection{迹公式和 $\theta$ 条件}
由前面指导 $s_{11}$ 和 $s_{22}$ 分别在 $D^{+}$ 和 $D^{-}$ 上解析, 其离散零点分别为 $ \{\xi_{n}\} $ 和 $ \{\xi_{n}^{*}\} $, 则
\begin{equation}
    \beta^{+} = s_{11}(z) \prod_{j=1}^{2N} \frac{z - \xi_{j}^{*}}{z - \xi_{j}}, \quad \beta^{-} = s_{22}(z) \prod_{j=1}^{2N} \frac{z - \xi_{j}}{z - \xi_{j}^{*}}.
    \label{ch5-eq-beta}
\end{equation}
仍在 $D^{+}$ 和 $D^{-}$ 上分别解析, 但不再有零点, 且 $\beta^{+} \beta^{-} = s_{11} s_{22} (z \in \sum)$. 由
\begin{equation}
    \det{S(z)} = s_{11}(z) s_{22}(z) - s_{12}(z) s_{21}(z) = 1 \implies \frac{1}{s_{11}(z) s_{22}(z)} = 1 - \rho(z) \tilde{\rho}(z) = 1 + \rho(z) \rho^{*}(z^{*}),
\end{equation}
则
\begin{equation}
    \beta^{+} \beta^{-} = s_{11} s_{22} = \frac{1}{1 + \rho(z) \rho^{*}(z^{*})} \implies \ln{\beta^{+}} - (- \ln{\beta^{-}}) = - \ln{(1 + \rho(z) \rho^{*}(z^{*}))}, \quad z \in \sum.
    \label{ch5-eq-beta-jump}
\end{equation}
由 Plemelj 公式可得
\begin{equation}
    \ln{\beta^{\pm}(z)} = \mp \frac{1}{2\pi \rmi} \int_{\sum} \frac{\ln{(1 + \rho(s) \rho^{*}(s^{*}))}}{s - z} \rmd s, \quad z \in D^{\pm}.
    \label{ch5-eq-beta-plemelj}
\end{equation}
将上式带入 \eqref{ch5-eq-beta} 可得迹公式
\begin{align}
    s_{11}(z) & = \exp{\left[ - \frac{1}{2 \pi \rmi} \int_{\sum} \frac{ln{(1+\rho(z) \rho^{*}(z^{*}))}}{s-z} \rmd s \right]} \prod_{j=1}^{2N} \frac{z - \xi_{j}}{z - \xi_{j}^{*}}, \quad z \in D^{+}, \label{ch5-eq-trace-s11} \\
    s_{22}(z) & = \exp{\left[ \frac{1}{2 \pi \rmi} \int_{\sum} \frac{ln{(1+\rho(z) \rho^{*}(z^{*}))}}{s-z} \rmd s \right]} \prod_{j=1}^{2N} \frac{z - \xi_{j}^{*}}{z - \xi_{j}}, \quad z \in D^{-}.
    \label{ch5-eq-trace-s22}
\end{align}
特别的, 无反射情形下, $s_{12} = s_{21} = 0$, 则上述迹公式可简化为
\begin{equation}
    s_{11} = \prod_{j=1}^{2N} \frac{z - \xi_{j}}{z - \xi_{j}^{*}}, \quad z \in D^{+}; \quad s_{22} = \prod_{j=1}^{2N} \frac{z - \xi_{j}^{*}}{z - \xi_{j}}, \quad z \in D^{-}.
\end{equation}
令 $z \to 0$ 则 \eqref{ch5-eq-trace-s11} 化为 (此时 $s_{11} = q_{-}/q_{+}$, 见 \eqref{ch5-eq-S-asymptotic})
\begin{equation}
    \frac{q_{-}}{q_{+}} = \exp{ \left[ \frac{\rmi}{2 \pi} \int_{\sum} \frac{\ln{(1+\rho(s) \rho^{*}(s^{*}))}}{s} \rmd s \right]} \exp{ \left[ 4 \rmi \sum_{j=1}^{2N} \arg(\xi_{j}) \right]}, z \in D^{+},
\end{equation}
因此可得 $\theta$ 条件
\begin{equation}
    \arg{\frac{q_{-}}{q_{+}}} = \frac{1}{2\pi} \int_{\sum} \frac{\ln{(1+\rho(s) \rho^{*}(s^{*}))}}{s} \rmd s + 4 \sum_{j=1}^{2N} \arg(\xi_{j}).
\end{equation}
特别的, 在无反射情形下, $\theta$ 条件化为
\begin{equation}
    \arg{\frac{q_{-}}{q_{+}}} = 4 \sum_{j=1}^{2N} \arg(\xi_{j}).
\end{equation}
\begin{remark}
    这里的 $\theta$ 条件 (相位差条件) 揭示了非零边界 NLS 方程中宏观边界与微观散射数据之间的物理一致性: 系统在空间两端 ($x \to \pm\infty$) 的背景相位差 $\Delta\theta = \theta_{-} - \theta_{+}= \arg(q_-/q_+)$, 为系统内部所有孤子(离散谱) 的相位贡献与色散辐射 (连续谱) 的相位贡献之和. 同样也说明暗孤子的存在，必然伴随着背景波的固定相位跳变, 没有相位差就暗孤子.

    (这里 $\Delta \theta$ (相位跳变, Phase shift) 为物理方面的叫法, 数学方面叫相位差( Phase difference))
\end{remark}
\subsection{无反射势情形}
在无反射情形下, $G=0$, 则 \eqref{ch5-eq-mu-12} 和 \eqref{ch5-eq-mu-21} 化为
\begin{align}
    \mu_{-,12}(x, t, \xi_{j})     & = - \frac{\rmi q_{-}}{\xi_{j}} - \sum_{k=1}^{2N} \frac{C_{k}^{*} e^{\rmi \theta(\xi_{k}^{*})} \mu_{-,11}(x, t, \xi_{k}^{*})}{\xi_{j} - \xi_{k}^{*}}, \label{ch5-eq-mu-12-sol} \\
    \mu_{-,11}(x, t, \xi_{j}^{*}) & = 1 + \sum_{k=1}^{2N} \frac{C_{k} e^{\rmi \theta(\xi_{k})} \mu_{-,12}(x, t, \xi_{k})}{\xi_{j}^{*} - \xi_{k}}, \quad j = 1, 2, \cdots, 2 N. \label{ch5-eq-mu-11-sol}           \\
    q(x,t)                        & = q_{-} + \sum_{n=1}^{2N} C_{n}^{*} e^{\rmi \theta(\xi_{n}^{*})} \mu_{-,11}(x, t, \xi_{n}^{*}). \label{ch5-eq-q}
\end{align}
为表示方便, 引入记号
\begin{equation}
    c_{j}(x, t, z) = \frac{C_{j}}{z - \xi_{j}} e^{-2\rmi \theta(z)}, \quad j = 1, 2, \cdots, 2 N,
\end{equation}
观察发现 $c_{k}^{*}(\xi_{j}^{*}) = C_{k}^{*}/(\xi_{j} - \xi_{k}^{*})$. 则 \eqref{ch5-eq-mu-12-sol} 和 \eqref{ch5-eq-mu-11-sol} 化为
\begin{align}
    \mu_{-,12}(\xi_{j})     & = \frac{\rmi q_{-}}{\xi_{j}} - \sum_{k=1}^{2N} c_{k}^{*}(\xi_{j}^{*}) \mu_{-,11}(\xi_{k}^{*}), \\
    \mu_{-,11}(\xi_{j}^{*}) & = 1 + \sum_{k=1}^{2N} c_{k}(\xi_{j}^{*}) \mu_{-,12}(\xi_{k}), \quad j = 1, 2, \cdots, 2 N.
\end{align}
联立上式得
\begin{equation}
    \mu_{-,11}(\xi_{n}^{*}) = 1 - \rmi q_{-} \sum_{j=1}^{2N} \frac{c_{j}(\xi_{n}^{*})}{\xi_{j}} - \sum_{j=1}^{2N} \sum_{k=1}^{2N} c_{j}(\xi_{n}) c_{k}^{*}(\xi_{j}^{*}) \mu_{-,11}(\xi_{k}^{*}), \quad n = 1, 2, \cdots, 2 N.
    \label{ch5-eq-mu11}
\end{equation}
这样由公式 \eqref{ch5-eq-q} 可知, 求解 NLS 方程解转化为求解上述线性方程组的 $\mu_{-,11}(\xi_{n}^{*})$, 我们引入适当记号即可将其写为矩阵形式.
\subsection{NLS 方程的 $N$ 孤子解}
引入记号
\begin{equation}
    X = (X_{1}, X_{2}, \cdots, X_{2N})^{T}, \quad B = (B_{1}, B_{2}, \cdots, B_{2N})^{T}, A = (A_{nk})
\end{equation}
其中
\begin{equation}
    X_{n} = \mu_{-,11}(\xi_{n}^{*}), \quad B_{n} = 1 - \rmi q_{-} \sum_{j=1}^{2N} \frac{c_{j}(\xi_{n}^{*})}{\xi_{j}}, \quad A_{nk} = \sum_{j=1}^{2N} c_{j}(\xi_{n}^{*}) c_{k}^{*}(\xi_{j}^{*}),
\end{equation}
则方程组 \eqref{ch5-eq-mu11} 改写为
\begin{equation}
    M X = B, \quad M = I + A = (M_{1}, \cdots, M_{2N}).
\end{equation}
由 Cramer 规则可得
\begin{equation}
    X_{n} = \frac{\det{M_{n}^{rep}}}{\det{M}}, \quad n = 1, 2, \cdots, 2 N,
\end{equation}
其中
\begin{equation}
    M_{n}^{rep} = (M_{1}, \cdots, M_{n-1}, B, M_{n+1}, \cdots, M_{2N}).
\end{equation}
因此 NLS 方程的 $N$ 孤子解可表示为
\begin{equation}
    \begin{aligned}
        q(x,t) & = q_{-} + \frac{\rmi}{\det{M}} \sum_{n=1}^{2N} Y_{n} \det{M_{n}^{rep}},                                                       \\
               & = q_{-} - \frac{\rmi}{\det{M}} \sum_{n=1}^{2N} (-1)^{1+n+1} Y_{n} \det{(B, M_{1}, \cdots, M_{n-1}, M_{n+1}, \cdots, M_{2N})}, \\
               & = q_{-} - \rmi \frac{\det{M^{aug}}}{\det{M}},
    \end{aligned}
\end{equation}
其中 $Y_{n} = C_{n}^{*} e^{\rmi \theta(\xi_{n}^{*})}$, $M^{aug}$ 为 $M$ 的增广矩阵
\begin{equation}
    M^{aug} = \begin{pmatrix}
        0 & Y \\ B & M
    \end{pmatrix}, \quad Y = (Y_{1}, Y_{2}, \cdots, Y_{2N}).
\end{equation}
